include(GNUInstallDirs)
set(HIFACE_DEFAULT_SDK "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
if(NOT EXISTS "${HIFACE_DEFAULT_SDK}" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtimes")
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" HIFACE_PROCESSOR)
    if(HIFACE_PROCESSOR MATCHES "^(aarch64|arm64)$")
        set(HIFACE_PROCESSOR arm64)
    elseif(HIFACE_PROCESSOR MATCHES "^(x86_64|amd64)$")
        set(HIFACE_PROCESSOR x86_64)
    else()
        message(FATAL_ERROR "Unsupported Linux processor: ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    set(HIFACE_DEFAULT_SDK "${CMAKE_CURRENT_SOURCE_DIR}/runtimes/${HIFACE_PROCESSOR}/sdk")
endif()
set(HIFACE_SDK_DIR "${HIFACE_DEFAULT_SDK}" CACHE PATH "Installed HiFace SDK directory")
if(NOT TARGET HiFace::HiFace)
    find_package(HiFace CONFIG REQUIRED HINTS "${HIFACE_SDK_DIR}")
endif()
if(HiFace_VERSION)
    set(hiface_demo_sdk_version "${HiFace_VERSION}")
else()
    set(hiface_demo_sdk_version "${HIFACE_LINUX_VERSION}")
endif()
if(NOT hiface_demo_sdk_version)
    message(FATAL_ERROR "The Demo requires the linked HiFace SDK version")
endif()
find_package(Threads REQUIRED)
set(HIFACE_DEMO_VERSION "${hiface_demo_sdk_version}" CACHE STRING "Demo application version")
function(hiface_harden_demo target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE
            "-include${CMAKE_CURRENT_SOURCE_DIR}/src/common/stack_protection.h"
            "$<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:-fstack-protector-strong>"
            "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=."
            "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=.")
        set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
        get_target_property(kind ${target} TYPE)
        if(kind STREQUAL "EXECUTABLE")
            target_link_options(${target} PRIVATE "LINKER:-z,relro,-z,now,-z,noexecstack" "-pie")
        endif()
    endif()
endfunction()
